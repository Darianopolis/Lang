CommaSeparatedList(T)
    Nothing
    T (',' T)* ','?

BinaryOperator
    '+'         -- Add
    '-'         -- Subtract
    '*'         -- Multiply
    '/'         -- Divide
    '<<'        -- Shift left
    '>>'        -- Shift right
    '^'         -- Bit-wise Xor assign
    '&'         -- Bit-wise And
    '|'         -- Bit-wise Or
    '**'        -- Exponent
    '%'         -- Modulo

    '+='        -- Add assign
    '-='        -- Subtract assign
    '*='        -- Multiply assign
    '/='        -- Divide assign
    '<<='       -- Shift left assign
    '>>='       -- Shift right assign
    '^='        -- Bit-wise Xor assign
    '&='        -- Bit-wise And assign
    '|='        -- Bit-wise Or assign
    '**='       -- Exponent assign
    '%='        -- Modulo assign

    '..'        -- Range (exclusive)
    '..='       -- Range (inclusive)

    '&&'        -- Logical And
    '||'        -- Logical Or

    '='         -- Assignment

InfixExpr
    Expr BinaryOperator Expr

Return
    'return' Expr

PrefixExpr
    '+' Expr    -- Arithmetic identity
    '-' Expr    -- Arithmetic negation
    '~' Expr    -- Bit-wise negation
    '!' Expr    -- Logical negation
    '++' Expr   -- Pre-increment
    '--' Expr   -- Pre-decrement

    Return

Argument
    Expr
    '.' Identifier '=' Expr

Arguments
    CommaSeparatedList(Argument)

Call
    Expr '(' Arguments ')'

Subscript
    Expr '[' Arguments ']'

FieldAccess
    Expr '.' Identifier

PostfixExpr
    Expr '++'   -- Post-increment
    Expr '--'   -- Post-decrement

    Call
    Subscript
    FieldAccess

Terminal
    Number
    String
    Identifier

LoopControlFlow
    'break'
    'continue'

VarDecl
    Identifier (':' Expr? ('=' Expr)?)?

TypeDecl
    ':' Expr ('=' Expr)?

Fields
    (VarDecl ';')*

Struct
    '@struct' '{' Fields '}'

Parameter
    VarDecl
    TypeDecl

Parameters
    CommaSeparatedList(Parameter)

FunctionPrototype
    '(' Parameters ')' ('->' Expr)?

Function
    FunctionPrototype Block

EnumValue
    Identifier ('=' Expr?)

EnumValues
    CommaSeparatedList(EnumValue)

Enum
    'enum' (': Expr)? '{' EnumValues '}'

Group
    '(' Expr ')'

PureExpr
    Terminal
    LoopControlFlow

    Struct
    Function
    Enum
    Group

    InfixExpr
    PrefixExpr
    PostfixExpr

Else
    'else' BlockExpr
    'else' If

If
    'if' Expr Block Else?

For
    'for' Expr '->' '(' Parameters ')' Block

While
    'while' Expr Block

Defer
    'defer' Block

Block
    '{' Statement* Expr? '}'

Expr
    PureExpr
    BlockExpr

BlockExpr
    If
    For
    While
    Block

Statement
    BlockExpr
    Defer
    VarDecl ';'
    PureExpr ';'
    ';'

File
    Statement*
